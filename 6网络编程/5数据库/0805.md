```sql
SELECT  -- 查询数据列
    c1,
    c2,
    c3
FROM 
    t1
OUTER/INNER JOIN    -- 来自于多张表
    t2
ON
    t1.id=t2.pk
WHERE   -- 原过滤
    c1 > 100 AND c2 < 9
GROUP BY    -- 分组
    c1
HAVING  -- 分组后的数据过滤
    c4 > 100
ORDER BY    -- 排序
    id
LIMIT   -- 限定数量
    3;
```

### 多表查询（表连接）
无效：笛卡尔乘积，行 n*m，列 n+m
    SELECT * FROM staff,dept;

- 内连接（等值连接）
- 外连接    inner join
    - 左外连接  left join
    - 右外连接  right join
    - 全连接    full join(不支持), 左外连接 union 右外连接

-- 内连接
```sql
select c1,c2
from t1
inner join t2
on t1.pk=t2.fk;

select s.id,s.name,d.name dept, d.city
from staff s
inner join dept d
on s.dept_id = d.id;
-- 简化版
select * from t1, t2 where t1.pk=t2.fk;

select s.id,s.name,d.name dept,city from staff s, dept d where dept_id=d.id;

```

-- 外连接
```sql
-- 左连接：join 左侧表中所有记录都出现
select s.id,s.name,d.name dept, d.city
from staff s
left join dept d
on s.dept_id = d.id;

-- 右连接：join 右侧表中所有记录都出现
select s.id,s.name,d.name dept, d.city
from staff s
right join dept d
on s.dept_id = d.id;

-- 全连接(左右相并)
select s.id,s.name,d.name dept, d.city
from staff s
left join dept d
on s.dept_id = d.id
union
select s.id,s.name,d.name dept, d.city
from staff s
right join dept d
on s.dept_id = d.id;

```

```sql
create table t1
(
    id int auto_increment primary key,
    val char(4)
);
create table t2
(
    id int auto_increment primary key,
    val char(4)
);
insert into t1(val) values('A'),('B'),('C');
insert into t2(val) values('C'),('X'),('Y');

-- 交集 inner join
select *
from t1
inner join t2
on t1.val=t2.val;

-- 并集 union
select val from t1
union 
select val from t2;

-- 并集，不合并重复项 union all
select val from t1
union all
select val from t2;

-- 差集
-- t1 - t2 not in 性能比较差
select val from t1 where val not in (select val from t2);
-- t2 - t1
select val from t2 where val not in (select val from t1);

-- 左连接 + is null
select * from t1 left join t2 on t1.val=t2.val where t2.id is NULL;
select * from t2 left join t1 on t1.val=t2.val where t1.id is NULL;

-- 研发部门的员工信息
select staff.*,dept.name,dept.city 
from staff 
join dept on dept_id=dept.id 
where dept.name="研发";
-- 或者
select * from staff
where dept_id = 
(select id from dept where name = "研发");
```

### 分组
- GROUP BY  分组
- HAVING    分组后过滤

```sql
select gender from staff group by gender;

select gender, count(id) from staff group by gender;

select level, count(id) from staff group by level;

select level, count(id), max(salary) max_s
from staff
group by level
having max_s > 10000
order by level desc;

-- 统计各个部门名称，人数平均工资
select d.name, count(s.id), avg(salary)
from dept d
left join staff s
on d.id = s.dept_id
group by d.name;

-- 统计各个城市人数，最低最高工资，以及人数最多的城市
select d.city, count(s.id) c,min(salary),max(salary)
from dept d
left join staff s
on d.id = s.dept_id
group by d.city
order by c desc
limit 2;

-- 统计员工以及对应老板
select s.id, s.name, b.name boss
from staff s
left join staff b
on s.boss=b.id;

select b.id, b.name boss, count(s.id) N
from staff s
right join staff b
on s.boss=b.id
group by b.id, boss
order by N desc;
```
### 模糊查询

使用 like 关键字，配合 _, %
    _ 匹配一个字符
    % 匹配任意个字符

select name from staff where name like '%a%';   包含字符a
select name from staff where name like '%';     第一个字符为a
select name from staff where name like '_a%';   第二个字符为a

### 子查询

在一个查询中嵌套其他查询

select * from staff
where salary > (select avg(salary) from staff);

select * from staff
where
dept_id = (select dept_id from staff where name= 'tom')
and
birthday > (select birthday from staff where name= 'tom');
