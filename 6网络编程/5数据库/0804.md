# 数据库 - SQL

## DDL 数据定义语言
- CREATE
- DROP
- ALTER
- GRANT
- TRUNCATE

```sql
create database obd;
use obd;
show tables;

create table t2
(
    id int auto_increment primary key,
    c2 char(4),
    c3 decimal(7,2)
);
-- 查看表结构
desc t2;
-- 修改表结构
ALTER TABLE t2 ADD c4 varchar(16) DEFAULT 'N';
ALTER TABLE t2 CHANGE c4 title varchar(26);
ALTER TABLE t2 DROP COLUMN c3;
ALTER TABLE t2 CHANGE title `order` VARCHAR(56);
-- 插入数据
INSERT INTO t2(`order`) VALUES('abcd');
INSERT INTO t2(`order`) VALUES('fgh');
INSERT INTO t2(`order`) VALUES('sde');
-- 查看数据
SELECT * FROM t2;

-- 截断表：先drop，在create
-- 删除了表的数据，释放了表原来的空间，重置了自动增长的序列
-- 没有删除表的结构
TRUNCATE TABLE t2;

-- 删除表(数据，结构，空间)
DROP TABLE t2;

-- 删除表(数据)/满足2222条件才删除，序列不会重置
DELETE FROM t2 WHERE 2222;
DELETE FROM t2;


-- 创建用户 abc 密码 123456
CREATE USER 'abc'@'%' IDENTIFIED BY '123456';
-- 给abc授予obd所有权限
GRANT ALL ON obd.* TO 'abc'@'%';
GRANT ALL ON mall.* TO 'abc'@'%';

-- 登录 abc
mysql -u abc -p123456


-- 启用远程连接
cd/etc/mysql/mysql.conf.d
sudo nano mysqld.cnf
-- 将绑定地址改为 0.0.0.0
-- 重启服务
sudo service mysql restart
sudo netstat -tpnl | grep mysql


```
