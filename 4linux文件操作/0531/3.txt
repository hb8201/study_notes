    EVP_DigestInit_ex(ctx, EVP_sha256(), NULL);      // 中间参数 EVP_md5 EVP_sha1 EVP_sha256
    EVP_DigestUpdate(ctx, str, strlen(str));
    int len;
    EVP_DigestFinal_ex(ctx, hash, &len);
    EVP_MD_CTX_free(ctx);                           // 释放空间
    printf("MD5:\t");
    for (int i = 0; i < len; i++)

        printf("%02x", hash[i]);
    }
    printf("\n");
}
