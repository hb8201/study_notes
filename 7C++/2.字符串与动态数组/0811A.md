```c
    for (auto r : s)        // r 为局部变量，拷贝 s 的每个值，只读

    for (auto &r : s)       // 引用，可以修改 s

    for (const auto &r : s) // 常量引用，不可修改 s，性能比 for (auto r : s) 好

    for (auto &&r : s)      // 万能引用，可以修改，临时变量可用
```
