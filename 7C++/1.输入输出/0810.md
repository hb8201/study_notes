```c
// 着重于函数调用
int date;
// 猫
struct type t;
// 叫（猫，喵喵）
void f1(struct type *t, int a);
// 吃（猫，猫粮）
void f2(struct type *t, int a, int b);


// 
// 猫
class Cls
{
    int date;
    // 叫
    void f1(){}
    //吃
    void f2(){}
}
// 猫 二狗
Cls obj;
// 二狗.叫（喵喵）
obj.f1(1);
// 二狗.吃（猫粮）
obj.f2(2,3);

```
