# 结构体数组、指针

## 结构体数组

```c
    // 定义常量，宏定义
#define NAME_SIZE 10

// 定义一个新的数据类型
typedef struct Student
{
    char name[NAME_SIZE];   // 姓名（成员、字段）
    int age;                // 年龄
} Student;


    Student a[4] = {
        {"吕布", 21}, 
        {"赵子龙", 19}, 
        {.age=23, .name="貂蝉"}
    };
    strncpy(a[3].name, "曹操", NAME_SIZE-1);
    a[3].age = 24;

    // 指针变量
    Student *p = a;
    for (int i = 0; i < 4; i++)
    {
        // printf("%s\t%d\n", a[i].name, a[i].age);     // 数组下标
        // printf("%s\t%d\n", (*p).name, (*p).age);     // 指针
        printf("%s\t%d\n", p->name, p->age);            // 指针 - >
        p++;
    }
```

```c
struct Date
{
    int year;
    int month;
    int day;
};

typedef struct Person
{
    char name[16];          // 姓名
    int age;                // 年龄
    struct Date birthday;   // 生日
    struct Person *lover;   // 配偶 指针
} Person;


int main(int argc, char const *argv[])
{
    Person p2 = {"bob"};
    Person p1 = {"alice", 21, {2004,1,13}, &p2};
    // p1.lover = &p2;

    printf("%s\t%d\t%d-%d-%d\t%s\n", p1.name, p1.age, 
        p1.birthday.year, 
        p1.birthday.month, 
        p1.birthday.day,
        p1.lover->name
    );

    return 0;
}
```
