### 分支语句
- `if ()`
- `if () else`
- `if () else if () else if () else`
- `switch() {case }`

```c
    // 条件
    int a;
    if (7 == a) // 1;

    if ()
        // 1;

    if ()
    {
        // 1
        // 2
    }
    // if - else 两个代码块
    if ()
    {
        // 1
    }
    else
    {
        // 2
    }

    // 3 多路分支
    if (1)
    {
        // 1
    }
    else if (2)
    {
        // 2
    }
    else if (3)

    }

        // 4
    }

```
